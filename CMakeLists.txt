cmake_minimum_required(VERSION 3.16)
project(real_sokoban LANGUAGES CXX)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport REQUIRED)
add_library(RealSokoban SHARED src/RealSokoban.cc)
target_compile_features(RealSokoban PRIVATE cxx_std_17)
target_link_libraries(RealSokoban PRIVATE gz-sim::gz-sim gz-plugin::gz-plugin gz-transport::gz-transport)
target_compile_options(RealSokoban PRIVATE -Wall -Wextra)
